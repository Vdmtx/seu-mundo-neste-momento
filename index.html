// Grade mais densa para melhor cobertura (10° de latitude/longitude) const points=[]; for(let lat=-80;lat<=80;lat+=10){ for(let lng=-170;lng<=170;lng+=10){ points.push({lat,lng}); } } const batchSize=30,heatData=[]; } return; } const lats=batch.map(p=>p.lat).join(','); const lngs=batch.map(p=>p.lng).join(','); try{ const res=await fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lats}&longitude=${lngs}¤t=temperature_2m&timezone=UTC&forecast_days=1`); const data=await res.json(); const items=Array.isArray(data)?data:[data]; items.forEach((item,idx)=>{ const temp=item.current?.temperature_2m; if(temp!==undefined && temp!==null){ // Normalizar: -40°C = 0.0, +50°C = 1.0 const intensity=Math.max(0,Math.min(1,(temp+40)/90)); heatData.push([batch[idx].lat,batch[idx].lng,intensity]); fetchedCount++; } }); document.getElementById('conn-status').textContent=`Temperatura: ${Math.round(fetchedCount/points.length*100)}%`; // Próximo batch com delay para rate limit await new Promise(r=>setTimeout(r,100)); fetchBatch(startIdx+batchSize); }catch(e){ console.warn('Erro ao buscar temperatura:',e); fetchBatch(startIdx+batchSize); } } function createHeatmap(data){ // Criar heatmap com parâmetros otimizados para gradientes suaves tempLayer=L.heatLayer(data,{ radius: 50, // Maior radius para criar sobreposição suave blur: 40, // Mais blur para transições suaves maxZoom: 5, // Limitar zoom máximo max: 1.0, minOpacity: 0.4, // Opacidade mínima para não ficar muito forte gradient: { 0.0: '#4a0080', // Roxo escuro (-40°C) 0.1: '#0000ff', // Azul (-30°C) 0.2: '#0066ff', // Azul claro (-20°C) 0.3: '#00ccff', // Ciano (-10°C) 0.4: '#00ffcc', // Verde-azulado (0°C) 0.5: '#00ff66', // Verde (10°C) 0.6: '#88ff00', // Verde-amarelo (15°C) 0.7: '#ffff00', // Amarelo (25°C) 0.8: '#ffcc00', // Amarelo-alaranjado (30°C) 0.9: '#ff8800', // Laranja (40°C) 1.0: '#ff0000' // Vermelho (50°C) } });
